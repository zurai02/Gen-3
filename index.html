Quick Start

Zero-config setup. One line to load, one line to create a window.

lua 
-- Load the library
local InfinityOmni = loadstring(game:HttpGet("https://raw.githubusercontent.com/zurai02/Gen-3/main/Main.lua"))()

-- Create a window
local Window = InfinityOmni.new({
    name = "My Script",
    theme = "dark",
    size = UDim2.new(0, 500, 0, 600),
    draggable = true,
    resizable = true,
})

-- Add a tab
local MainTab = Window:addTab("Main")

-- Add components
Window:addButton({name = "Click Me", callback = function() print("clicked") end}, MainTab)
Window:addToggle({name = "Auto Farm", default = false, callback = function(v) print(v) end}, MainTab)
Window:addSlider({name = "Speed", min = 0, max = 100, default = 50}, MainTab)
Window:addInput({name = "Username", placeholder = "Enter name..."}, MainTab)

lua
-- Access any module directly
local Signal = InfinityOmni.Signal
local ThemeManager = InfinityOmni.ThemeManager
local StateManager = InfinityOmni.StateManager

-- Create standalone signals
local mySignal = Signal.new()
mySignal:Connect(function(data) print(data) end)
mySignal:Fire("hello")

-- Manage state independently
local state = StateManager.new()
state:setState("health", 100)
state:watch("health", function(new, old)
    print("Health changed from", old, "to", new)
end)

-- Switch themes at runtime
Window.theme:setTheme("light")
Window.theme:customizeColor("accent", Color3.fromRGB(255, 0, 100))

Advanced Patterns

Build complex interactions using the exposed modules.

lua
-- Reactive UI: Auto-update button text when state changes
local state = Window.stateManager
state:setState("coins", 0)

local coinBtn = Window:addButton({name = "Coins: 0"}, MainTab)
state:watch("coins", function(newVal)
    coinBtn.instance.TextLabel.Text = "Coins: " .. newVal
end)

-- Theme-aware custom component
local function CreateCard(parent, title, theme)
    local card = Instance.new("Frame")
    card.Size = UDim2.new(1, -20, 0, 80)
    card.BackgroundColor3 = theme.colors.surfaceAlt
    card.Parent = parent

    theme.changed:Connect(function(colors)
        card.BackgroundColor3 = colors.surfaceAlt
    end)

    return card
end

-- Animation chaining
local anim = Window.animationEngine
anim:tween(frame, 0.3, {Position = UDim2.new(0, 100, 0, 0)}).Completed:Connect(function()
    anim:tween(frame, 0.2, {Size = UDim2.new(0, 200, 0, 200)})
end)
